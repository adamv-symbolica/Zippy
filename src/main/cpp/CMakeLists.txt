cmake_minimum_required(VERSION 3.22)
project(zippy_openfst LANGUAGES CXX)
find_package(OpenFst CONFIG QUIET)
add_library(zippy_openfst SHARED zippy_openfst.cpp)
target_compile_features(zippy_openfst PUBLIC cxx_std_17)
target_include_directories(zippy_openfst PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(TARGET openfst::fst)
  # Upstream installs headers under openfst/lib; derive their prefix from the
  # selected package rather than accidentally picking a distro header tree.
  get_filename_component(OPENFST_PREFIX "${OpenFst_DIR}/../../.." ABSOLUTE)
  find_path(OPENFST_INCLUDE_ROOT NAMES openfst/lib/fstlib.h fst/fstlib.h
    PATHS "${OPENFST_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
  target_include_directories(zippy_openfst PRIVATE ${OPENFST_INCLUDE_ROOT})
  if(NOT EXISTS "${OPENFST_INCLUDE_ROOT}/openfst/lib/fstlib.h")
    target_compile_definitions(zippy_openfst PRIVATE ZIPPY_OPENFST_LEGACY=1)
  endif()
  target_link_libraries(zippy_openfst PUBLIC openfst::fst)
else()
  find_path(FST_INCLUDE_DIR NAMES fst/fstlib.h REQUIRED)
  find_library(FST_LIBRARY NAMES fst REQUIRED)
  target_include_directories(zippy_openfst PRIVATE ${FST_INCLUDE_DIR})
  target_compile_definitions(zippy_openfst PRIVATE ZIPPY_OPENFST_LEGACY=1)
  target_link_libraries(zippy_openfst PUBLIC ${FST_LIBRARY})
endif()
set(ZIPPY_PROGRAM "" CACHE FILEPATH "Generated C++ source to compile as a standalone program")
if(ZIPPY_PROGRAM)
  add_executable(zippy_program ${ZIPPY_PROGRAM})
  target_link_libraries(zippy_program PRIVATE zippy_openfst)
endif()
option(ZIPPY_BUILD_BENCHMARKS "Build native construction and algebra microbenchmarks" OFF)
if(ZIPPY_BUILD_BENCHMARKS)
  add_executable(zippy_openfst_benchmark openfst_benchmark.cpp)
  target_link_libraries(zippy_openfst_benchmark PRIVATE zippy_openfst)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(zippy_openfst_test openfst_runtime_test.cpp)
  target_link_libraries(zippy_openfst_test PRIVATE zippy_openfst)
  add_test(NAME zippy_openfst_runtime COMMAND zippy_openfst_test)
endif()
